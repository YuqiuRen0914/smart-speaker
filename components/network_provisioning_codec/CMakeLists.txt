file(READ "${CMAKE_CURRENT_LIST_DIR}/UPSTREAM.json" identity)
string(JSON count LENGTH "${identity}" files)
math(EXPR last "${count} - 1")
foreach(index RANGE ${last})
  string(JSON relative MEMBER "${identity}" files ${index})
  string(JSON expected GET "${identity}" files "${relative}")
  file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/upstream/${relative}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Official provisioning original changed: ${relative}")
  endif()
endforeach()

  INCLUDE_DIRS upstream/include upstream/proto-c REQUIRES esp_netif protocomm protobuf-c PRIV_REQUIRES log)
